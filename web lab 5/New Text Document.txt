  <script type="text/babel">
    const { useState, useEffect } = React;

    function App() {
      const [movies, setMovies] = useState(() => {
        const saved = localStorage.getItem("movieWatchList");
        return saved ? JSON.parse(saved) : [];
      });

      const [title, setTitle] = useState("");
      const [rating, setRating] = useState(5);
      const [comment, setComment] = useState("");

      useEffect(() => {
        localStorage.setItem("movieWatchList", JSON.stringify(movies));
      }, [movies]);

      const handleAddMovie = (e) => {
        e.preventDefault();
        if (!title.trim()) return;

        const newMovie = {
          id: Date.now(),
          title: title.trim(),
          rating: Number(rating),
          comment: comment.trim(),
        };

        setMovies([newMovie, ...movies]);
        setTitle("");
        setRating(5);
        setComment("");
      };

      const handleRemoveMovie = (id) => {
        setMovies(movies.filter((movie) => movie.id !== id));
      };

      return (
        <div className="app-container">
          <header>
            <h1>🎬 Movies Watch List</h1>
            <p>Track the movies you watch, rate them, and leave a review.</p>
          </header>

          <form className="movie-form" onSubmit={handleAddMovie}>
            <div className="form-group">
              <label htmlFor="title">Movie Title</label>
              <input
                id="title"
                type="text"
                placeholder="e.g., Interstellar"
                value={title}
                onChange={(e) => setTitle(e.target.value)}
                required
              />
            </div>

            <div className="form-group">
              <label htmlFor="rating">Star Rating (1 - 5)</label>
              <select
                id="rating"
                value={rating}
                onChange={(e) => setRating(Number(e.target.value))}
              >
                <option value={5}>5 Stars (⭐⭐⭐⭐⭐)</option>
                <option value={4}>4 Stars (⭐⭐⭐⭐)</option>
                <option value={3}>3 Stars (⭐⭐⭐)</option>
                <option value={2}>2 Stars (⭐⭐)</option>
                <option value={1}>1 Star (⭐)</option>
              </select>
            </div>

            <div className="form-group">
              <label htmlFor="comment">Comment / Review</label>
              <textarea
                id="comment"
                rows={3}
                placeholder="Write your thoughts on the movie..."
                value={comment}
                onChange={(e) => setComment(e.target.value)}
              />
            </div>

            <button type="submit" className="btn-add">
              + Add Movie
            </button>
          </form>

          <section className="movie-list">
            <h2>Watched Movies ({movies.length})</h2>

            {movies.length === 0 ? (
              <p className="empty-state">No movies added yet. Add one above!</p>
            ) : (
              movies.map((movie) => (
                <div key={movie.id} className="movie-card">
                  <div className="movie-header">
                    <h3>{movie.title}</h3>
                    <span className="star-display" title={`${movie.rating} out of 5`}>
                      {"⭐".repeat(movie.rating)}
                    </span>
                  </div>

                  {movie.comment && (
                    <p className="movie-comment">"{movie.comment}"</p>
                  )}

                  <button
                    className="btn-remove"
                    onClick={() => handleRemoveMovie(movie.id)}
                  >
                    Remove
                  </button>
                </div>
              ))
            )}
          </section>
        </div>
      );
    }

    const root = ReactDOM.createRoot(document.getElementById("root"));
    root.render(<App />);
  </script>
